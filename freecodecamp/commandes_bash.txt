Aide-Mémoire Commandes Bash / Linux

## 1. NAVIGATION & RÉPERTOIRES
  cd [dossier]      Change de répertoire (cd / cd ~ = dossier personnel, cd .. = dossier parent)
  ls                Liste le contenu du répertoire courant
  ls -l             Affichage détaillé (permissions, taille, propriétaire, date)
  ls -a             Affiche tous les fichiers, y compris les fichiers cachés (.nom_fichier)
  ls -la            Combine le format détaillé (-l) et l'affichage des fichiers cachés (-a)

## 2. GESTION DES FICHIERS & DOSSIERS
  mkdir <dossier>    Crée un nouveau répertoire
  rmdir <dossier>    Supprime un répertoire vide
  touch <fichier>    Crée un fichier vide ou met à jour sa date de modification
  rm <fichier>        Supprime un fichier
  rm -r <dossier>    Supprime un dossier et tout son contenu de façon récursive
  cp <src> <dest>    Copie un fichier vers une destination
  cp -r <src> <dest> Copie un dossier et tout son contenu
  mv <src> <dest>    Déplace ou renomme un fichier/dossier
                       - Renommer : mv ancien.txt nouveau.txt
                       - Déplacer : mv fichier.txt /dossier/

## 3. LECTURE, ÉDITION, FLUX & REDIRECTIONS
  cat <fichier>      Affiche la totalité du contenu d'un fichier dans la console
  more <fichier>     Affiche le contenu d'un fichier page par page (Espace pour défiler)
  nano <fichier>     Ouvre un éditeur de texte interactif dans le terminal
  echo "<texte>"     Affiche le texte dans le terminal
  > ou 1>            Redirige la sortie standard (stdout) et ÉCRASE le fichier cible
  >> ou 1>>          Redirige la sortie standard (stdout) et AJOUTE à la fin du fichier
  2>                 Redirige la sortie d'erreur (stderr) vers un fichier
  <                  Redirige l'entrée standard (stdin) depuis un fichier
  | (pipe)           Passe la sortie d'une commande en entrée à une autre (ex: cmd1 | cmd2)

## 4. RECHERCHE, COMPTAGE & TRAITEMENT DE TEXTE
  find <dossier> -name "<nom>"
                       Recherche des fichiers par nom dans une arborescence
                       - Dans 'src' : find src -name "app.js"
                       - Tous les .txt : find . -name "*.txt"

  wc <fichier>       Compte les lignes, mots et octets/caractères d'un fichier
                       - wc -l : Compte uniquement les lignes
                       - wc -w : Compte uniquement les mots
                       - wc -m ou -c : Compte les caractères / octets

  grep '<motif>' <fichier>
                       Cherche un texte ou motif dans un fichier
                       - -i : Ignore la casse (majuscules/minuscules)
                       - -n : Affiche le numéro des lignes trouvées
                       - -v : Inverse la recherche (affiche les lignes ne contenant pas le motif)
                       - -c : Compte le nombre de lignes correspondantes
                       - -o : Extrait et affiche uniquement la partie exacte qui correspond au motif
                       - --color : Surligne en couleur le motif trouvé dans la console

  sed 's/ancien/nouveau/' <fichier>
                       Éditeur de flux pour substituer ou modifier du texte
                       - sed 's/r/2/' f.txt : Remplace la 1ère occurrence de chaque ligne
                       - sed 's/r/2/g' f.txt : Remplace toutes les occurrences (substitution globale)
                       - sed -E 's/([0-9]+).*/\1/' : Utilise les regex étendues (ex: groupes (), +)
                       - sed -i 's/a/b/g' f.txt : Modifie directement le fichier d'origine (in-place)
                       - sed -i '3d' f.txt : Supprime la 3ème ligne du fichier
                       - sed -i '/erreur/d' f.txt : Supprime les lignes contenant 'erreur'
                       - sed -i '2i texte' f.txt : Insère du texte à la ligne 2

## 5. EXÉCUTION DE SCRIPTS & PERMISSIONS
  which bash              Affiche le chemin vers l'exécutable Bash
  #!/usr/bin/bash         Shebang à placer en 1ère ligne du script
  chmod +x file.sh        Rend le script exécutable
  sh file.sh / bash file.sh / ./file.sh   Méthodes d'exécution d'un script

## 6. VARIABLES & ENTRÉES / SORTIES
  read Name                Lit une saisie utilisateur (read VAR < file.txt lit la 1ère ligne)
  $* ou $@                 Ensemble des arguments passés au script
  $?                       Code de retour de la dernière commande (0 = succès, non-zéro = erreur)
  [[ 4 -ge 5 ]]; echo $?  Affiche le statut d'exécution
  [[ -a file.sh ]]; echo $? Vérifie l'existence d'un fichier (0 si existe)
  ( I=6; J=7 )            Affectation de variables
  declare -p J            Affiche le type et la valeur de la variable J

## 7. STRUCTURES CONDITIONNELLES (if / elif / else)
  if [[ CONDITION ]]
  then
    STATEMENTS
  elif [[ CONDITION ]]
  then
    STATEMENTS
  else
    STATEMENTS
  fi

  if (( CONDITION ))
  then
    STATEMENTS
  fi

  Remarques sur les conditions :
  - [[ ... ]] utilise (-eq, -ne, -lt, -le, -gt, -ge)
  - (( ... )) utilise (>, <, >=, <=, ==, !=)
  - Options de comparaison entiers/fichiers :
      -eq (égal), -ne (différent), -lt (inférieur), -le (inférieur/égal), -gt (supérieur), -ge (supérieur/égal)
      -a (fichier existe), -x (fichier exécutable)
  - Opérateurs logiques : && (ET), || (OU)

## 8. BOUCLES & PAUSES
  for (( i = 10; i > 0; i-- ))
  do
    echo $i
  done

  while [[ CONDITION ]]
  do
    STATEMENTS
  done

  until [[ CONDITION ]]
  do
    STATEMENTS
  done

  sleep 3                 Fait une pause de 3 secondes

## 9. CALCULS & NOMBRES ALÉATOIRES
  (( ... ))                Effectue un calcul ou une opération sans rien afficher
  $(( ... ))               Remplace le calcul par son résultat
  NUMBER=$RANDOM          Nombre aléatoire (0 à 32767)
  NUMBER=$(( RANDOM % 75 + 1 )) Nombre aléatoire entre 1 et 75

## 10. TABLEAUX & FONCTIONS
  ARR=("a" "b" "c")
  echo ${ARR[1]}            Affiche le 2ème élément ("b")
  echo ${ARR[*]} / echo ${ARR[@]}  Affiche tous les éléments du tableau

  FUNCTION_NAME() {
    STATEMENTS
  }
